// ============================================================ // APPS SCRIPT - Plataforma Administrativa RCR // ============================================================ const SHEET_COLABORADORES = "Colaboradores"; const SHEET_SEGUIMIENTO = "Seguimiento"; const SHEET_SEMAFORO = "Semáforo"; const SHEET_USUARIOS = "Usuarios"; // ─── ENTRY POINT GET ──────────────────────────────────────── function doGet(e) { try { const action = e.parameter.action; const sheet = e.parameter.sheet; if (action === "getAll") return respond(getAll(sheet)); if (action === "getByName") return respond(getByName(sheet, e.parameter.nombre)); if (action === "login") return respond(login(e.parameter.correo, e.parameter.googleToken)); if (action === "getHeaders") return respond(getHeaders(sheet)); if (action === "getSheets") return respond(getSheets()); return respond({ success: false, message: "Acción no reconocida" }); } catch (err) { return respond({ success: false, message: err.message }); } } // ─── ENTRY POINT POST ─────────────────────────────────────── function doPost(e) { try { const body = JSON.parse(e.postData.contents); const action = body.action; const sheet = body.sheet; const data = body.data; if (action === "createRow" || action === "create") return respond(createRow(sheet, data)); if (action === "updateRow" || action === "update") return respond(updateRow(sheet, data, body.identifier)); if (action === "deleteRow" || action === "delete") return respond(deleteRow(sheet, body.rowId || (data && data.RowID), body.field)); if (action === "uploadPhoto") return respond(uploadPhoto(body.base64, body.fileName, body.colaborador)); if (action === "addColumn") return respond(addColumn(body.sheet, body.columnName, body.columnType, body.options)); if (action === "renameColumn") return respond(renameColumn(body.sheet, body.oldName, body.newName)); if (action === "deleteColumn") return respond(deleteColumn(body.sheet, body.columnName)); if (action === "reorderColumns") return respond(reorderColumns(body.sheet, body.order)); if (action === "updateColumnType") return respond(updateColumnType(body.sheet, body.columnName, body.columnType, body.options)); return respond({ success: false, message: "Acción no reconocida" }); } catch (err) { return respond({ success: false, message: err.message }); } } // ─── OBTENER TODOS ────────────────────────────────────────── function getAll(sheetName) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(sheetName); if (!sheet) return { success: false, message: "Hoja no encontrada: " + sheetName }; const values = sheet.getDataRange().getValues(); if (values.length < 2) return { success: true, data: [] }; const headers = values[0]; const rows = values.slice(1).filter(r => r[0] !== "").map(row => { const obj = {}; headers.forEach((h, i) => obj[h] = row[i] !== undefined ? String(row[i]) : ""); return obj; }); return { success: true, data: rows }; } // ─── OBTENER POR NOMBRE ───────────────────────────────────── function getByName(sheetName, nombre) { const result = getAll(sheetName); if (!result.success) return result; return { success: true, data: result.data.filter(row => row["Colaborador"] === nombre) }; } // ─── OBTENER HOJAS ────────────────────────────────────────── function getSheets() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheets = ss.getSheets().map(s => s.getName()); return { success: true, data: sheets }; } // ─── OBTENER COLUMNAS CON TIPOS ───────────────────────────── function getHeaders(sheetName) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(sheetName); if (!sheet) return { success: false, message: "Hoja no encontrada: " + sheetName }; const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0]; // Leer metadatos de tipos guardados en PropertiesService const props = PropertiesService.getScriptProperties(); const metaKey = 'colTypes_' + sheetName; let columnMeta = {}; try { columnMeta = JSON.parse(props.getProperty(metaKey) || '{}'); } catch(e) {} const columns = headers.filter(h => h !== '').map((h, i) => ({ name: String(h), index: i, type: columnMeta[h]?.type || 'texto', options: columnMeta[h]?.options || [] })); return { success: true, data: columns }; } // ─── AGREGAR COLUMNA ──────────────────────────────────────── function addColumn(sheetName, columnName, columnType, options) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(sheetName); if (!sheet) return { success: false, message: "Hoja no encontrada: " + sheetName }; // Verificar que no exista ya const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0]; if (headers.includes(columnName)) return { success: false, message: "Ya existe una columna con ese nombre" }; // Agregar al final const newCol = sheet.getLastColumn() + 1; sheet.getRange(1, newCol).setValue(columnName); // Guardar tipo en PropertiesService saveColumnMeta(sheetName, columnName, columnType, options || []); // Si es desplegable u opción múltiple, agregar validación de datos if ((columnType === 'desplegable' || columnType === 'multiple') && options && options.length > 0) { const lastRow = Math.max(sheet.getLastRow(), 2); const range = sheet.getRange(2, newCol, lastRow - 1, 1); const rule = SpreadsheetApp.newDataValidation() .requireValueInList(options, true) .setAllowInvalid(false) .build(); range.setDataValidation(rule); } return { success: true, message: "Columna agregada: " + columnName }; } // ─── RENOMBRAR COLUMNA ────────────────────────────────────── function renameColumn(sheetName, oldName, newName) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(sheetName); if (!sheet) return { success: false, message: "Hoja no encontrada: " + sheetName }; const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0]; const idx = headers.indexOf(oldName); if (idx < 0) return { success: false, message: "Columna no encontrada: " + oldName }; // Verificar que no exista otra con el nuevo nombre if (headers.includes(newName)) return { success: false, message: "Ya existe una columna con ese nombre" }; sheet.getRange(1, idx + 1).setValue(newName); // Actualizar metadatos const props = PropertiesService.getScriptProperties(); const metaKey = 'colTypes_' + sheetName; let meta = {}; try { meta = JSON.parse(props.getProperty(metaKey) || '{}'); } catch(e) {} if (meta[oldName]) { meta[newName] = meta[oldName]; delete meta[oldName]; props.setProperty(metaKey, JSON.stringify(meta)); } return { success: true }; } // ─── ELIMINAR COLUMNA ─────────────────────────────────────── function deleteColumn(sheetName, columnName) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(sheetName); if (!sheet) return { success: false, message: "Hoja no encontrada: " + sheetName }; const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0]; const idx = headers.indexOf(columnName); if (idx < 0) return { success: false, message: "Columna no encontrada: " + columnName }; sheet.deleteColumn(idx + 1); // Limpiar metadatos const props = PropertiesService.getScriptProperties(); const metaKey = 'colTypes_' + sheetName; let meta = {}; try { meta = JSON.parse(props.getProperty(metaKey) || '{}'); } catch(e) {} delete meta[columnName]; props.setProperty(metaKey, JSON.stringify(meta)); return { success: true }; } // ─── REORDENAR COLUMNAS ───────────────────────────────────── function reorderColumns(sheetName, newOrder) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(sheetName); if (!sheet) return { success: false, message: "Hoja no encontrada: " + sheetName }; const lastRow = sheet.getLastRow(); const lastCol = sheet.getLastColumn(); const allData = sheet.getRange(1, 1, lastRow, lastCol).getValues(); const headers = allData[0]; // Construir nuevo orden de datos const newData = allData.map(row => { return newOrder.map(colName => { const i = headers.indexOf(colName); return i >= 0 ? row[i] : ''; }); }); // Limpiar y reescribir sheet.clearContents(); sheet.getRange(1, 1, newData.length, newData[0].length).setValues(newData); return { success: true }; } // ─── ACTUALIZAR TIPO DE COLUMNA ───────────────────────────── function updateColumnType(sheetName, columnName, columnType, options) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(sheetName); if (!sheet) return { success: false, message: "Hoja no encontrada: " + sheetName }; const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0]; const idx = headers.indexOf(columnName); if (idx < 0) return { success: false, message: "Columna no encontrada: " + columnName }; // Guardar tipo saveColumnMeta(sheetName, columnName, columnType, options || []); // Actualizar validación de datos const lastRow = sheet.getLastRow(); const range = sheet.getRange(2, idx + 1, Math.max(lastRow - 1, 1), 1); range.clearDataValidations(); if ((columnType === 'desplegable' || columnType === 'multiple') && options && options.length > 0) { const rule = SpreadsheetApp.newDataValidation() .requireValueInList(options, true) .setAllowInvalid(false) .build(); range.setDataValidation(rule); } else if (columnType === 'numero') { const rule = SpreadsheetApp.newDataValidation() .requireNumber() .setAllowInvalid(true) .build(); range.setDataValidation(rule); } else if (columnType === 'fecha') { const rule = SpreadsheetApp.newDataValidation() .requireDate() .setAllowInvalid(true) .build(); range.setDataValidation(rule); } return { success: true }; } // ─── HELPER: GUARDAR METADATOS ────────────────────────────── function saveColumnMeta(sheetName, columnName, columnType, options) { const props = PropertiesService.getScriptProperties(); const metaKey = 'colTypes_' + sheetName; let meta = {}; try { meta = JSON.parse(props.getProperty(metaKey) || '{}'); } catch(e) {} meta[columnName] = { type: columnType, options: options || [] }; props.setProperty(metaKey, JSON.stringify(meta)); } // ─── LOGIN ────────────────────────────────────────────────── function login(correo, googleToken) { if (!correo) return { success: false, message: "Correo requerido" }; const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(SHEET_USUARIOS); if (!sheet || sheet.getDataRange().getValues().length < 2) { if (correo === "comunicacion@redesconrostro.org") return { success: true, nombre: "Administrador", rol: "Admin", correo }; return { success: false, message: "No hay usuarios registrados" }; } const values = sheet.getDataRange().getValues(); const headers = values[0]; const rows = values.slice(1); const iCorreo = headers.findIndex(h => String(h).toLowerCase().includes("correo")); const iNombre = headers.findIndex(h => String(h).toLowerCase().includes("nombre")); const iRol = headers.findIndex(h => String(h).toLowerCase().includes("rol")); const iActivo = headers.findIndex(h => String(h).toLowerCase().includes("activo")); for (const row of rows) { if (String(row[iCorreo]||"").toLowerCase().trim() === correo.toLowerCase().trim()) { if (String(row[iActivo]||"true").toLowerCase() === "false") return { success: false, message: "Usuario inactivo" }; return { success: true, correo, nombre: String(row[iNombre]||correo), rol: String(row[iRol]||"Visualizador") }; } } if (correo === "comunicacion@redesconrostro.org") return { success: true, nombre: "Administrador", rol: "Admin", correo }; return { success: false, message: "Usuario no autorizado." }; } // ─── CREAR ────────────────────────────────────────────────── function createRow(sheetName, data) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(sheetName); if (!sheet) return { success: false, message: "Hoja no encontrada: " + sheetName }; const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0]; if (headers.includes("RowID") && !data["RowID"]) { const allRows = sheet.getDataRange().getValues(); const lastId = allRows.length > 1 ? parseInt(allRows[allRows.length-1][0]) || 0 : 0; data["RowID"] = lastId + 1; } sheet.appendRow(headers.map(h => data[h] !== undefined ? data[h] : "")); return { success: true, RowID: data["RowID"] || "" }; } // ─── ACTUALIZAR ───────────────────────────────────────────── function updateRow(sheetName, data, identifier) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(sheetName); if (!sheet) return { success: false, message: "Hoja no encontrada: " + sheetName }; const values = sheet.getDataRange().getValues(); const headers = values[0]; const keyCol = headers.includes("RowID") ? "RowID" : (headers.includes("Colaborador") ? "Colaborador" : headers[0]); const keyVal = identifier || data[keyCol]; const keyIdx = headers.indexOf(keyCol); for (let i = 1; i < values.length; i++) { if (String(values[i][keyIdx]) === String(keyVal)) { headers.forEach((h, j) => { if (data[h] !== undefined) sheet.getRange(i+1, j+1).setValue(data[h]); }); return { success: true }; } } return { success: false, message: "Registro no encontrado" }; } // ─── ELIMINAR ─────────────────────────────────────────────── function deleteRow(sheetName, rowId, field) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(sheetName); if (!sheet) return { success: false, message: "Hoja no encontrada: " + sheetName }; const values = sheet.getDataRange().getValues(); const headers = values[0]; const searchField = field || (headers.includes("RowID") ? "RowID" : "Colaborador"); const searchIdx = headers.indexOf(searchField); if (searchIdx < 0) return { success: false, message: "Campo no encontrado: " + searchField }; for (let i = 1; i < values.length; i++) { if (String(values[i][searchIdx]) === String(rowId)) { sheet.deleteRow(i + 1); return { success: true }; } } return { success: false, message: "Registro no encontrado" }; } // ─── SUBIR FOTO ───────────────────────────────────────────── function uploadPhoto(base64Data, fileName, colaboradorNombre) { try { const decoded = Utilities.base64Decode(base64Data); const mimeType = (fileName||'').toLowerCase().endsWith('.png') ? 'image/png' : 'image/jpeg'; const blob = Utilities.newBlob(decoded, mimeType, colaboradorNombre + '_foto'); let folder; const query = "mimeType='application/vnd.google-apps.folder' and title='Fotos RCR - Colaboradores' and trashed=false"; const results = DriveApp.searchFolders(query); if (results.hasNext()) { folder = results.next(); } else { folder = DriveApp.getRootFolder().createSubfolder('Fotos RCR - Colaboradores'); } const existing = folder.getFilesByName(colaboradorNombre + '_foto'); while (existing.hasNext()) existing.next().setTrashed(true); const file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); const fileId = file.getId(); const fileUrl = 'https://drive.google.com/thumbnail?id=' + fileId + '&sz=w300'; const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(SHEET_COLABORADORES); if (sheet) { const values = sheet.getDataRange().getValues(); const headers = values[0]; const colFoto = headers.indexOf('Foto'); const colName = headers.indexOf('Colaborador'); if (colFoto >= 0 && colName >= 0) { for (let i = 1; i < values.length; i++) { if (String(values[i][colName]) === String(colaboradorNombre)) { sheet.getRange(i+1, colFoto+1).setValue(fileUrl); break; } } } } return { success: true, url: fileUrl, fileId }; } catch (err) { return { success: false, message: err.toString() }; } } // ─── TEST ──────────────────────────────────────────────────── function testDrive() { const token = ScriptApp.getOAuthToken(); Logger.log('Token OK: ' + token.substring(0,20) + '...'); } // ─── RESPUESTA ─────────────────────────────────────────────── function respond(data) { return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON); }
